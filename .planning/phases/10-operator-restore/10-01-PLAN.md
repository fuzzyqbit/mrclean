// src/state/session-map.ts — ADDITIVE export beside V2_TOKEN_RE (line 308).
// Unanchored global twin: same capture groups, `g` flag, no ^$ anchors.
export const V2_TOKEN_SCAN_RE: RegExp // /<MRCLEAN:([A-Z0-9_]+):(\d{3}|OVF):([a-f0-9]{8})>/g

// src/restore/index.ts — pure module: no node:fs, no node:crypto, no config input.
export interface RestoreResult {
  text: string
  restored: number         // exact index hits replaced
  unmatched: number        // unknown/stale/planted/OVF tokens passed through
  skippedSecret: number    // tokens present in secretPlaceholders passed through
  restoredOriginals: string[] // one entry PER restored occurrence, scan order — raw values,
                              // memory-only; the CLI (10-05) dedupes+hashes at its boundary
}
export function restoreText(
  input: string,
  index: ReadonlyMap<string, string>,          // placeholder -> original (policy-gated by 10-02)
  secretPlaceholders?: ReadonlySet<string>,     // placeholders of secret-class map entries
): RestoreResult

<feature>
  <name>Single-pass exact-lookup restore engine</name>
  <files>src/restore/index.ts, src/state/session-map.ts, tests/restore/engine.test.ts</files>
  <behavior>
    One `input.replace(V2_TOKEN_SCAN_RE, callback)` pass. Callback order: (1) OVF label short-circuit BEFORE any lookup — return token, unmatched++; (2) secretPlaceholders membership — return token, skippedSecret++; (3) exact full-token `index.get(token)` — hit: return original, restored++, push to restoredOriginals; miss: return token, unmatched++.

    Cases (given index { '&lt;MRCLEAN:WORD:001:aabbccdd&gt;' -> 'proj-zeus', '&lt;MRCLEAN:PII_EMAIL:003:aabbccdd&gt;' -> 'kim@zz.invalid' } and secretPlaceholders { '&lt;MRCLEAN:AWS_KEY:002:aabbccdd&gt;' }):
    - exact hit => replaced with 'proj-zeus'; restored=1; restoredOriginals=['proj-zeus']
    - wrong nonce '&lt;MRCLEAN:WORD:001:ffffffff&gt;' => unchanged, unmatched=1 (planted-token defense)
    - unissued NNN '&lt;MRCLEAN:WORD:999:aabbccdd&gt;' => unchanged, unmatched=1 (enumeration defense)
    - fabricated TYPE '&lt;MRCLEAN:BOGUS_T:001:aabbccdd&gt;' => unchanged, unmatched=1
    - v1 token '&lt;MRCLEAN:WORD:001&gt;' (no nonce) => never matches the scan regex; text unchanged; ALL counters 0
    - OVF token '&lt;MRCLEAN:WORD:OVF:aabbccdd&gt;' => unchanged, unmatched=1 — EVEN when the index is hand-poisoned with that OVF token as a key (belt-and-braces: label short-circuit precedes lookup, Pitfall 1)
    - secret placeholder => unchanged, skippedSecret=1, NOT in restoredOriginals
    - no-cascade (Pitfall 2): index maps tokenA -> literal text of tokenB (tokenB also an index key). Input containing tokenA and tokenB separately => tokenA's occurrence becomes tokenB-text and STAYS tokenB-text; the separate literal tokenB occurrence becomes B's original; restored=2, no third replacement
    - mixed document with all classes => counters sum correctly; per-occurrence: same token twice => restored=2, restoredOriginals has 2 entries
    - empty input => { text: '', restored: 0, unmatched: 0, skippedSecret: 0 }
    - token-free input => byte-identical text, all counters 0
    - immutability: the passed-in index Map and secretPlaceholders Set are not mutated (size + entries identical after call)
    - grammar-sync lock: V2_TOKEN_RE.source === '^' + V2_TOKEN_SCAN_RE.source + '$' AND V2_TOKEN_SCAN_RE.flags === 'g' (drift between allocator and restorer fails loudly)
  </behavior>
  <implementation>
    RESEARCH Code Example 2 is the target shape, extended with the secretPlaceholders parameter. Import V2_TOKEN_SCAN_RE from '../state/session-map.js' — never re-declare the grammar (Don't-Hand-Roll table). No mutation (build counters locally, return a new object). File under 200 lines.
  </implementation>
