```typescript
// Frozen partition (D-11 / RESEARCH Pattern 5). Everything NOT in this list is
// secret-class, INCLUDING unknown future TYPEs (fail-toward-privacy).
export const RESTORABLE_TYPES: readonly string[] // WORD, PII_EMAIL, PII_PHONE, PII_IP, PII_PERSON, PII_ORG, PII_LOC
export function isRestorableType(type: string): boolean

// Discriminated entry shapes — SecretMapEntry has NO `original` at the type level.
export interface SecretMapEntry { placeholder: string; type: string; counter: number }
export interface RestorableMapEntry { placeholder: string; type: string; counter: number; original: string }
export type SessionMapEntry = SecretMapEntry | RestorableMapEntry

export interface SessionMapV1 {
  version: 1
  sessionId: string
  nonce8: string       // 8 lowercase hex chars — per-session CSPRNG token tag (D-10)
  hashSalt: string     // 64 lowercase hex chars — 32 random bytes for HMAC addressing
  counter: number      // authoritative NNN counter (grows only)
  entries: Record<string, SessionMapEntry>  // keyed by hmacAddress(hashSalt, original)
}

export function createEmptySessionMap(sessionId: string): SessionMapV1
export function makeMapEntry(type: string, placeholder: string, counter: number, original: string): SessionMapEntry
export function serializeSessionMap(map: SessionMapV1): string   // strips `original` from non-restorable entries (defense in depth)
export function parseSessionMap(json: string): SessionMapV1 | null // hand guards, null on any shape failure
export function hmacAddress(saltHex: string, value: string): string // HMAC-SHA256 hex (64 chars)
export function formatV2Token(type: string, counter: number, nonce8: string): string
  // counter <= 999 => `<MRCLEAN:TYPE:NNN:nonce8>` (3-digit zero-pad); counter > 999 => `<MRCLEAN:TYPE:OVF:nonce8>`
export const V2_TOKEN_RE: RegExp // /^<MRCLEAN:([A-Z0-9_]+):(\d{3}|OVF):([a-f0-9]{8})>$/

export const SESSION_ID_RE: RegExp // /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i (Pitfall 5)
export function isValidSessionId(sid: string): boolean

// Hydration contract consumed by PlaceholderManager (09-04, type-only import)
// and produced by the state facade (09-05). Closures keep manager free of state runtime imports.
export interface ReversibleHydration {
  nonce8: string
  counterFloor: number
  entriesByHmac: ReadonlyMap<string, { placeholder: string; type: string }>
  hmacOf: (value: string) => string
  formatToken: (type: string, counter: number) => string
}
export interface PendingAllocation {
  value: string                  // raw original — memory only, never logged
  type: string
  provisionalPlaceholder: string
}
export interface PlaceholderRename { from: string; to: string }
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-09-02-01 | Information disclosure | makeMapEntry / serializeSessionMap | mitigate | Secret-class entries structurally lack `original` (type level) AND serializer strips it (defense in depth); unknown TYPE => secret-class; no config parameter exists in the module |
| T-09-02-02 | Information disclosure | hmacAddress | mitigate | HMAC-SHA256 with per-session 32-byte random salt (stored only inside the encrypted envelope) — a leaked decrypted map does not permit dictionary guess-confirmation of secret originals, unlike plain sha256 |
| T-09-02-03 | Information disclosure | v2 tokens | mitigate | formatV2Token embeds per-session CSPRNG nonce8 — kills planted-token enumeration (D-10) |
| T-09-02-04 | Elevation of privilege | SESSION_ID_RE | mitigate | Strict UUID allowlist (never sanitize-by-transform); 'mcp-server' and traversal strings fail structurally |
| T-09-02-05 | Information disclosure | HMAC collision | accept | ~2^-128 per pair => two originals share a placeholder => cosmetic wrong-value restore in Phase 10; documented in code comment per RESEARCH Pattern 3 |
